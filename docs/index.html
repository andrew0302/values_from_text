make_apa_freq_table <- function(df, bin_col,
                                title = "Table",
                                subtitle = "Frequencies by bin") {

  bin_sym   <- enquo(bin_col)
  bin_label <- as_label(bin_sym)   # for column header

  # 1) counts within levels of factor
  sum_df <- df %>%
    filter(selected %in% c("keep","remove", "used")) %>%
    mutate(selected = factor(selected, levels = c("remove","keep", "used"))) %>%
    count(!!bin_sym, selected, name = "n") %>%
    # ensure both levels exist for every bin
    tidyr::complete(!!bin_sym, selected, fill = list(n = 0)) %>%
    group_by(!!bin_sym) %>%
    summarise(
      Total        = sum(n),
      Removed_n    = sum(n[selected == "remove"]),
      Kept_n       = sum(n[selected == "keep"]),
      Annotated_n  = sum(n[selected == "used"]),
      Removed_pct  = Removed_n / Total,
      Kept_pct     = Kept_n / Total,
      Annotated_pct = Annotated_n / Total, 
      .groups = "drop"
    ) %>%
    arrange(desc(Removed_n)) %>%
    relocate(Total, .after = last_col())

  # 2) build gt table
  # Note: cols_merge() will merge the *_n and *_pct columns into a single display column.
  gt_tab <- sum_df %>%
    gt() %>%
    
    # define title and subtitle
    tab_header(title = title, subtitle = subtitle) %>%
    
    # center column headers
    tab_style(
      style = cell_text(align = "center"),
      locations = cells_column_labels(columns = everything())) |>
    
    # headers of each column
    cols_label(
      !!bin_sym   := bin_label,
      Total       = html("Total n"),
      Removed_n   = "Removed",
      Removed_pct = "Removed %",
      Kept_n      = "Kept",
      Kept_pct    = "Kept %", 
      Annotated_n = "Annotated", 
      Annotated_pct = "Annotated %"
    ) %>%
    
    # set number of decimal places
    fmt_number(columns = Total, decimals = 0) %>%
    fmt_number(columns = c(Removed_n, Kept_n, Annotated_n), decimals = 0) %>%
    fmt_percent(columns = c(Removed_pct, Kept_pct, Annotated_pct), decimals = 1) %>%
    
    # merge n and % into "n (xx.x%)"
    cols_merge(
      columns = c(Removed_n, Removed_pct),
      pattern = "{1} ({2})") %>%
    cols_merge(
      columns = c(Kept_n, Kept_pct),
      pattern = "{1} ({2})") %>%
    cols_merge(
      columns = c(Annotated_n, Annotated_pct),
      pattern = "{1} ({2})") %>%
    cols_hide(columns = c(Removed_pct, Kept_pct, Annotated_pct)) %>%
    
    # set which lines show
    tab_options(
      table.border.top.width = px(0),
      table.border.bottom.width = px(0),
      
      # bold under the heading (title/subtitle)
      heading.border.bottom.width = px(2),
      
      # remove body row lines
      table_body.hlines.width = px(0),
      table_body.border.top.width = px(0),
      table_body.border.bottom.width = px(0),
      
      column_labels.border.top.width = px(0),
      column_labels.border.bottom.width = px(1)
    )

  gt_tab
}

Popularity

Popularity was extremely exponentially distributed, with an extreme left skew. We manually binned to make the quantiles per each of the 7 bins as similar as possible. Thus, the first bin contained the lowest 40% of the population in terms of popularity, while the 7th bin contained the highest 9%:

  • Bin 0: 1–2 playlist appearances (≤ 40%)
  • Bin 2: 2–3 appearances (41–53%)
  • Bin 3: 3–4 appearances (54–60%)
  • Bin 4: 4–6 appearances (61–70%)
  • Bin 5: 7–14 appearances (71–80%)
  • Bin 6: 15–60 appearances (81–91%)
  • Bin 7: 61–203,345 appearances (top 9% of artists; 91–100%)

# function to summarize by party
summarise_party <- function(data) {
  data %>%
    group_by(party) %>%
    summarise(
      n_excerpts    = n(),
      n_speeches    = n_distinct(url),
      n_presidents  = n_distinct(president),
      .groups = "drop"
    ) %>%
    mutate(
      excerpts_per_president = n_excerpts / n_presidents,
      pct_excerpts = n_excerpts / sum(n_excerpts))}

# 1) Full pool

  participant_summary <- df |>
    group_by(participant_ID) |>     
    summarise(n_ratings = n()) |> 
    summarize(mean_n_ratings = mean(n_ratings), 
              sd_n_ratings = sd(n_ratings)) |> 
    ungroup()
    
  list(
    total_ratings = total_ratings,
    item_summary  = item_summary, 
    participant_summary = participant_summary
  )

# takes a correlation matrix and computes MDS coordinates
mds_data <- function(correlation_matrix){
  
  #compute distance matrix from correlation matrix
  idiss <- sim2diss(correlation_matrix)
  rownames(idiss) <- colnames(idiss)
  # fit ratio mds 
  fit_ratio <-  smacof::mds(idiss)
  
  # extract mds values and place in dataframe
  mds.values <- fit_ratio$conf
  mds.data   <- data.frame(
    Sample = rownames(mds.values), 
    X=mds.values[,1],
    Y=mds.values[,2], 
    # point stress contribution per value:
    # the larger the value, the less certain the estimate
    spp = fit_ratio$spp)
  
  return(mds.data)
}

rotate_mds <- function(df, degrees = 0) {
  radians <- degrees * pi / 180
  rot <- matrix(c(cos(radians), -sin(radians),
                  sin(radians),  cos(radians)), ncol = 2)
  rotated <- as.matrix(df[, c("X", "Y")]) %*% rot
  df$X <- rotated[, 1]
  df$Y <- rotated[, 2]
  df
}

# Nakagawa pseudo-R2 and AIC for a single model, as a one-row tibble.
model_fit_row <- function(model, name) {
  r2 <- as.data.frame(performance::r2_nakagawa(model))
  tibble::tibble(
    Model          = name,
    AIC            = AIC(model),
    R2_marginal    = r2$R2_marginal,
    R2_conditional = r2$R2_conditional)
}

# LRT of a larger model against a reference (nested) model.
# Returns chisq, df, and formatted p for the larger model.
lrt_row <- function(reference, larger) {
  a <- suppressWarnings(anova(reference, larger))
  p <- a$`Pr(>Chisq)`[2]
  p_fmt <- if (is.na(p)) NA_character_
           else if (p < 0.0001) "< .0001"
           else if (p < 0.001)  "< .001"
           else sprintf("= %.3f", p)
  tibble::tibble(
    LRT_chisq = a$Chisq[2],
    LRT_df    = a$Df[2],
    LRT_p     = p_fmt)
}

extract_spec_pretty <- function(model) {
  f_raw     <- paste(deparse(model@call$formula, width.cutoff = 500), collapse = " ")

  if (split_pos == -1) {

  } else {
    fixed  <- trimws(substr(f_raw, 1, split_pos - 1))
    random <- trimws(substr(f_raw, split_pos, nchar(f_raw)))
  }
  random <- gsub("\\s*\\+\\s*", " + ", random)

  html(paste0(fixed, "<br>", random))
}

build_anova_gt <- function(anova_obj,
                           eta_df,
                           title = "Type III ANOVA Results") {

  # Format ANOVA table
  anova_df <- anova_obj |>
    as.data.frame() |>
    tibble::rownames_to_column("Effect") |>
    mutate(Effect = trimws(Effect))

  # Format eta_df
  eta_df <- eta_df |>
    select(Effect, Eta2_partial) |>
    mutate(Effect = trimws(Effect)) |>
    distinct()

  # 3. Join
  df <- left_join(anova_df, eta_df, by = "Effect")

  # 4. Format p-values
  df <- df |>
    mutate(
      p_fmt = case_when(
        is.na(`Pr(>F)`)  ~ NA_character_,
        `Pr(>F)` < .0001 ~ "p < .0001",
        `Pr(>F)` < .001  ~ "p < .001",
        TRUE             ~ sprintf("p = %.3f", `Pr(>F)`)),

      F_fmt = sprintf("F(%d, %s) = %.2f",
                      NumDF,
                      format(round(DenDF), scientific = FALSE),
                      `F value`),

      eta_fmt = sprintf("%.3f", Eta2_partial))

  # 5. Group effects
  df <- df |>
    mutate(
      Group = case_when(
        !str_detect(Effect, ":") ~ "Main Effects",
        str_count(Effect, ":") == 1 ~ "Two-Way Interactions",
        str_count(Effect, ":") == 2 ~ "Three-Way Interactions",
        TRUE ~ "Four-Way Interactions"))

  # 6. Build gt table
  gt_tbl <- df |>
    select(Group, Effect, F_fmt, p_fmt, eta_fmt) |>      # CI removed
    gt(groupname_col = "Group") |>
    cols_label(
      Effect  = "Effect",
      F_fmt   = md("$F$"),
      p_fmt   = md("$p$"),
      eta_fmt = md("$\\eta^2_{partial}$")) |>
    cols_align("center", columns = c(F_fmt, p_fmt, eta_fmt)) |>
    tab_style(
    style = cell_text(align = "center"),
    locations = cells_row_groups()) |>
  tab_header(title = title)

  gt_tbl
}

build_anova_gt <- function(anova_obj, eta_df, title = "Type III ANOVA Results") {

  anova_df <- anova_obj |>
    as.data.frame() |>
    tibble::rownames_to_column("Effect") |>
    mutate(Effect = trimws(Effect))

  eta_df <- eta_df |>
    select(Effect, Eta2_partial) |>
    mutate(Effect = trimws(Effect)) |>
    distinct()

  df <- left_join(anova_df, eta_df, by = "Effect") |>
    mutate(
      p_fmt = case_when(
        is.na(`Pr(>F)`)  ~ NA_character_,
        `Pr(>F)` < .0001 ~ "p < .0001",
        `Pr(>F)` < .001  ~ "p < .001",
        TRUE             ~ sprintf("p = %.3f", `Pr(>F)`)
      ),
      F_fmt   = sprintf("F(%d, %s) = %.2f", NumDF,
                        format(round(DenDF), scientific = FALSE), `F value`),
      eta_fmt = sprintf("%.3f", Eta2_partial),
      Group = case_when(
        !str_detect(Effect, ":") ~ "Main Effects",
        str_count(Effect, ":") == 1 ~ "Two-Way Interactions",
        str_count(Effect, ":") == 2 ~ "Three-Way Interactions",
        TRUE ~ "Four-Way Interactions"
      )
    )

  df |>
    select(Group, Effect, F_fmt, p_fmt, eta_fmt) |>
    gt(groupname_col = "Group") |>
    cols_label(
      Effect  = "Effect",
      F_fmt   = md("$F$"),
      p_fmt   = md("$p$"),
      eta_fmt = md("$\\eta^2_{partial}$")
    ) |>
    cols_align("center", columns = c(F_fmt, p_fmt, eta_fmt)) |>
    tab_style(style = cell_text(align = "center"), locations = cells_row_groups()) |>
    tab_header(title = title)
}

# APA sentence helper: pairwise contrasts with z-tests and Cohen's d
apa_emm_pairs <- function(model, specs, total_sd, digits = 2) {

  cmp <- emmeans::emmeans(model, specs, lmer.df = "asymptotic") |>
    emmeans::contrast("pairwise", adjust = "tukey") |>
    broom::tidy()

  cmp |>
    dplyr::mutate(
      est   = sprintf(paste0("%.", digits, "f"), estimate),
      se    = sprintf(paste0("%.", digits, "f"), std.error),
      z_fmt = sprintf(paste0("%.", digits, "f"), statistic),
      p     = dplyr::case_when(
        adj.p.value < .001 ~ "< .001",
        TRUE ~ sprintf("= %.3f", adj.p.value)
      ),
      d_fmt = sprintf(paste0("%.", digits, "f"), estimate / total_sd),
      apa   = sprintf(
        "%s: difference = %s, SE = %s, z = %s, p %s, d = %s.",
        contrast, est, se, z_fmt, p, d_fmt)
    ) |>
    dplyr::pull(apa)
}

run_rra_by_group <- function(df) {

  df <- df |>
    # merge ethnicity and spectrum into a single variable
    mutate(group = paste(Ethnicity, spectrum, sep = " · "))

  bind_rows(
    # top
    df |>
      group_by(group) |>
      group_modify(~ {
        if (nrow(.x) < 2) return(tibble())
        
        rra <- aggregateRanks(.x$ranking, method = "RRA", exact = FALSE)
        rra |>
          rename(value = Name) |>
          mutate(direction = "top")}) |>
          ungroup(),

    # bottom
    df |>
      group_by(group) |>
      group_modify(~ {

        if (nrow(.x) < 2) return(tibble())

        rra <- aggregateRanks(lapply(.x$ranking, rev), 
                              method = "RRA",exact = FALSE)
        rra |>
          rename(value = Name) |>
          mutate(direction = "bottom")}) |>
        ungroup()
)}
